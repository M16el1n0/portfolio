bot.py — точка входа

bot.py 
import logging
from telegram import Update
from telegram.ext import (
    Application, CommandHandler, CallbackQueryHandler,
    PreCheckoutQueryHandler, MessageHandler, filters
)
from config import BOT_TOKEN
from handlers import start, catalog, payment, orders, admin
from database import init_db

logging.basicConfig(
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
    level=logging.INFO
)

async def main():
    await init_db()

    app = Application.builder().token(BOT_TOKEN).build()

    # Основные команды
    app.add_handler(CommandHandler("start", start.cmd_start))
    app.add_handler(CommandHandler("admin", admin.cmd_admin))

    # Каллбэки inline-кнопок
    app.add_handler(CallbackQueryHandler(catalog.show_plans, pattern=r"^catalog$"))
    app.add_handler(CallbackQueryHandler(payment.buy_plan, pattern=r"^buy_\d+$"))
    app.add_handler(CallbackQueryHandler(orders.my_orders, pattern=r"^orders$"))
    app.add_handler(CallbackQueryHandler(orders.resend_file, pattern=r"^resend_\d+$"))
    app.add_handler(CallbackQueryHandler(admin.admin_stats, pattern=r"^adm_stats$"))
    app.add_handler(CallbackQueryHandler(admin.admin_upload, pattern=r"^adm_upload_\d+$"))

    # Платёжные хэндлеры (Stars)
    app.add_handler(PreCheckoutQueryHandler(payment.pre_checkout))
    app.add_handler(MessageHandler(filters.SUCCESSFUL_PAYMENT, payment.successful_payment))

    # Загрузка файлов от админа
    app.add_handler(MessageHandler(filters.Document.ALL & filters.ChatType.PRIVATE, admin.handle_file_upload))

    await app.run_polling(drop_pending_updates=True)

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())

handlers/payment.py — логика оплаты

handlers/payment.py 
from telegram import Update, LabeledPrice
from telegram.ext import ContextTypes
from database import get_plan, save_order, get_file_for_plan, mark_file_sent
from config import ADMIN_ID

PLANS = {
    1: {"name": "Стартовый", "reviews": 50, "stars": 99},
    2: {"name": "Стандарт", "reviews": 150, "stars": 249},
    3: {"name": "Профи", "reviews": 400, "stars": 599},
}

async def buy_plan(update: Update, ctx: ContextTypes) -> None:
    query = update.callback_query
    await query.answer()

    plan_id = int(query.data.split("_")[1])
    plan = PLANS.get(plan_id)
    if not plan:
        await query.message.reply_text("Тариф не найден.")
        return

    await ctx.bot.send_invoice(
        chat_id=query.from_user.id,
        title=f"Отзывы — тариф «{plan['name']}»",
        description=f"{plan['reviews']} уникальных текстов отзывов в формате TXT",
        payload=f"plan_{plan_id}",
        currency="XTR",  # Telegram Stars
        prices=[LabeledPrice(plan["name"], plan["stars"])],
    )

async def pre_checkout(update: Update, ctx: ContextTypes) -> None:
    query = update.pre_checkout_query
    plan_id = int(query.invoice_payload.split("_")[1])
    file = await get_file_for_plan(plan_id)

    if not file:
        await query.answer(ok=False, error_message="Файлы временно недоступны. Попробуйте позже.")
    else:
        await query.answer(ok=True)

async def successful_payment(update: Update, ctx: ContextTypes) -> None:
    payment = update.message.successful_payment
    user_id = update.message.from_user.id
    plan_id = int(payment.invoice_payload.split("_")[1])

    file = await get_file_for_plan(plan_id)
    order_id = await save_order(user_id, plan_id, payment.telegram_payment_charge_id)

    await update.message.reply_document(
        document=file["file_id"],
        caption=f"✅ Оплата прошла!\n\n"
                f"Тариф: {PLANS[plan_id]['name']}\n"
                f"Отзывов: {PLANS[plan_id]['reviews']} шт.\n\n"
                f"Заказ №{order_id} сохранён в /orders"
    )
    await mark_file_sent(file["id"])

database.py — работа с БД

database.py 
import aiosqlite

DB_PATH = "reviews_bot.db"

async def init_db():
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            CREATE TABLE IF NOT EXISTS orders (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER NOT NULL,
                plan_id INTEGER NOT NULL,
                charge_id TEXT UNIQUE,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )""")
        await db.execute("""
            CREATE TABLE IF NOT EXISTS files (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                plan_id INTEGER NOT NULL,
                file_id TEXT NOT NULL,
                is_sent INTEGER DEFAULT 0,
                uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )""")
        await db.commit()

async def get_file_for_plan(plan_id: int):
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute(
            "SELECT * FROM files WHERE plan_id = ? AND is_sent = 0 ORDER BY id LIMIT 1",
            (plan_id,)
        ) as cur:
            return await cur.fetchone()

async def save_order(user_id: int, plan_id: int, charge_id: str) -> int:
    async with aiosqlite.connect(DB_PATH) as db:
        cur = await db.execute(
            "INSERT INTO orders (user_id, plan_id, charge_id) VALUES (?,?,?)",
            (user_id, plan_id, charge_id)
        )
        await db.commit()
        return cur.lastrowid

async def mark_file_sent(file_id: int):
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("UPDATE files SET is_sent = 1 WHERE id = ?", (file_id,))
        await db.commit()

async def get_user_orders(user_id: int):
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute(
            "SELECT * FROM orders WHERE user_id = ? ORDER BY created_at DESC LIMIT 10",
            (user_id,)
        ) as cur:
            return await cur.fetchall()
